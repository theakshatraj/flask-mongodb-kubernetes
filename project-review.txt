========================================

from flask import Flask, request, jsonify
from pymongo import MongoClient
from datetime import datetime
import os

# Initialize Flask application
app = Flask(__name__)

# Read MongoDB connection string from environment
MONGODB_URI = os.environ.get(
    "MONGODB_URI",
    "mongodb://localhost:27017/"
)

# Connect to MongoDB
client = MongoClient(
    MONGODB_URI,
    serverSelectionTimeoutMS=5000
)

# Select database and collection
db = client.flask_db
collection = db.data


# Root endpoint
@app.route("/", methods=["GET"])
def index():
    return (
        f"Welcome to the Flask app! "
        f"The current time is: {datetime.now()}"
    )


# Data endpoint
@app.route("/data", methods=["GET", "POST"])
def data():

    if request.method == "POST":

        document = request.get_json(silent=True)

        if not isinstance(document, dict):
            return jsonify({
                "error": "A valid JSON object is required"
            }), 400

        collection.insert_one(document)

        return jsonify({
            "status": "Data inserted"
        }), 201

    elif request.method == "GET":

        documents = list(
            collection.find({}, {"_id": 0})
        )

        return jsonify(documents), 200


# Run Flask application
if __name__ == "__main__":
    app.run(
        host="0.0.0.0",
        port=5000,
        debug=True
    )

# Use Python 3.10 slim image
FROM python:3.10-slim

# Set working directory
WORKDIR /app

# Prevent Python from generating bytecode
ENV PYTHONDONTWRITEBYTECODE=1

# Ensure logs appear immediately
ENV PYTHONUNBUFFERED=1

# Copy dependency file
COPY requirements.txt .

# Install dependencies
RUN pip install --no-cache-dir -r requirements.txt

# Copy application code
COPY app.py .

# Expose Flask application port
EXPOSE 5000

# Run Flask using Gunicorn
CMD ["gunicorn", "--bind", "0.0.0.0:5000", "--workers", "2", "app:app"]

kind: Deployment

  name: flask-app
  namespace: flask-mongodb

  replicas: 2

    matchLabels:
      app: flask-app

    metadata:
      labels:
        app: flask-app
    spec:
      containers:
        - name: flask-app
          image: theakshatraj/flask-mongodb-kubernetes:v1
          imagePullPolicy: IfNotPresent
          ports:
            - containerPort: 5000
              name: http
          env:
            - name: MONGODB_URI
              valueFrom:
                secretKeyRef:
                  name: mongodb-app-secret
                  key: MONGODB_URI

            requests:
              cpu: "200m"
              memory: "250Mi"

              cpu: "500m"
              memory: "500Mi"

              path: /
              port: 5000
            initialDelaySeconds: 5
            periodSeconds: 10
          livenessProbe:
            httpGet:
              path: /
              port: 5000

apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: mongodb
  namespace: flask-mongodb

spec:
  serviceName: mongodb-headless
  replicas: 1

  selector:
    matchLabels:
      app: mongodb

  template:
    metadata:
      labels:
        app: mongodb

    spec:
      containers:
        - name: mongodb
          image: mongo:6
          imagePullPolicy: IfNotPresent

          args:
            - mongod
            - --bind_ip_all
            - --auth

          ports:
            - name: mongodb
              containerPort: 27017
              protocol: TCP

          env:
            - name: MONGO_INITDB_ROOT_USERNAME
              valueFrom:
                secretKeyRef:
                  name: mongodb-secret
                  key: username

            - name: MONGO_INITDB_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: mongodb-secret
                  key: password

          resources:
            requests:
              cpu: "200m"
              memory: "250Mi"
            limits:
              cpu: "500m"
              memory: "500Mi"

          # Allows MongoDB sufficient time to start.
          startupProbe:
            tcpSocket:
              port: 27017
            periodSeconds: 5
            failureThreshold: 30
            timeoutSeconds: 3

          # Checks whether MongoDB is ready to accept connections.
          readinessProbe:
            tcpSocket:
              port: 27017
            periodSeconds: 10
            timeoutSeconds: 3
            failureThreshold: 3

          # Checks whether the MongoDB process is alive.
          livenessProbe:
            tcpSocket:
              port: 27017
            periodSeconds: 20
            timeoutSeconds: 3
            failureThreshold: 3

          volumeMounts:
            - name: mongodb-storage
              mountPath: /data/db

      volumes:
        - name: mongodb-storage
          persistentVolumeClaim:
            claimName: mongodb-pvc
